cmake_minimum_required(VERSION 3.19)
project(SpectralAudioDeltaForceVST2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SPECTRAL_MOCK "Build mock core only for local ABI tests; NEVER ship this build" OFF)
set(SPECTRAL_CORE_DIR "${CMAKE_SOURCE_DIR}/spectral-audio" CACHE PATH "Original Spectral Audio repository root")

if (SPECTRAL_MOCK)
  add_library(sa_core STATIC tests/mock_spectral.cpp)
  target_include_directories(sa_core PUBLIC tests)
else()
  foreach(f src/predictor.cpp src/model_registry.cpp src/stream.cpp generated/model_data.cpp include/spectral_cpu.h)
    if(NOT EXISTS "${SPECTRAL_CORE_DIR}/${f}")
      message(FATAL_ERROR "Original spectral-audio source missing: ${SPECTRAL_CORE_DIR}/${f}. Run build_plugin.cmd or clone original repo.")
    endif()
  endforeach()
  add_library(sa_core STATIC
    "${SPECTRAL_CORE_DIR}/src/predictor.cpp"
    "${SPECTRAL_CORE_DIR}/src/model_registry.cpp"
    "${SPECTRAL_CORE_DIR}/src/stream.cpp"
    "${SPECTRAL_CORE_DIR}/generated/model_data.cpp")
  target_include_directories(sa_core PUBLIC
    "${SPECTRAL_CORE_DIR}/include" "${SPECTRAL_CORE_DIR}/src" "${SPECTRAL_CORE_DIR}/generated")
endif()
set_target_properties(sa_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set_property(TARGET sa_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(sa_core PRIVATE /O2 /fp:strict /utf-8)
else()
  target_compile_options(sa_core PRIVATE -fno-fast-math -ffp-contract=off)
endif()

add_library(SpectralAudio_DeltaForce SHARED src/vst_plugin.cpp)
target_link_libraries(SpectralAudio_DeltaForce PRIVATE sa_core)
set_target_properties(SpectralAudio_DeltaForce PROPERTIES PREFIX "")
if(WIN32)
  target_sources(SpectralAudio_DeltaForce PRIVATE src/exports.def)
endif()
if(MSVC)
  set_property(TARGET SpectralAudio_DeltaForce PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(SpectralAudio_DeltaForce PRIVATE /O2 /fp:strict /utf-8)
endif()

include(CTest)
if(BUILD_TESTING AND SPECTRAL_MOCK AND UNIX)
  add_executable(vst_mock_host_test tests/vst_host_test.cpp)
  target_link_libraries(vst_mock_host_test PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME plugin_abi_and_delay COMMAND vst_mock_host_test $<TARGET_FILE:SpectralAudio_DeltaForce>)
endif()

if(BUILD_TESTING AND WIN32 AND NOT SPECTRAL_MOCK)
  add_executable(vst_win_smoke tests/vst_host_windows.cpp)
  add_test(NAME win_real_model_vst2_smoke COMMAND vst_win_smoke $<TARGET_FILE:SpectralAudio_DeltaForce>)
endif()
